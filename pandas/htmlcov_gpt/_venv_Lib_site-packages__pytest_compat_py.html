107def getfuncargnames( 

108 function: Callable[..., object], 

109 *, 

110 name: str = "", 

111 cls: type | None = None, 

112) -> tuple[str, ...]: 

113 """Return the names of a function's mandatory arguments. 

114 

115 Should return the names of all function arguments that: 

116 * Aren't bound to an instance or type as in instance or class methods. 

117 * Don't have default values. 

118 * Aren't bound with functools.partial. 

119 * Aren't replaced with mocks. 

120 

121 The cls arguments indicate that the function should be treated as a bound 

122 method even though it's not unless the function is a static method. 

123 

124 The name parameter should be the original name in which the function was collected. 

125 """ 

126 # TODO(RonnyPfannschmidt): This function should be refactored when we 

127 # revisit fixtures. The fixture mechanism should ask the node for 

128 # the fixture names, and not try to obtain directly from the 

129 # function object well after collection has occurred. 

130 

131 # The parameters attribute of a Signature object contains an 

132 # ordered mapping of parameter names to Parameter instances. This 

133 # creates a tuple of the names of the parameters that don't have 

134 # defaults. 

135 try: 

136 parameters = signature(function).parameters.values() 

137 except (ValueError, TypeError) as e: 

138 from _pytest.outcomes import fail 

139 

140 fail( 

141 f"Could not determine arguments of {function!r}: {e}", 

142 pytrace=False, 

143 ) 

144 

145 arg_names = tuple( 

146 p.name 

147 for p in parameters 

148 if ( 

149 p.kind is Parameter.POSITIONAL_OR_KEYWORD 

150 or p.kind is Parameter.KEYWORD_ONLY 

151 ) 

152 and p.default is Parameter.empty 

153 ) 

154 if not name: 

155 name = function.__name__ 

156 

157 # If this function should be treated as a bound method even though 

158 # it's passed as an unbound method or function, and its first parameter 

159 # wasn't defined as positional only, remove the first parameter name. 

160 if not any(p.kind is Parameter.POSITIONAL_ONLY for p in parameters) and ( 

161 # Not using `getattr` because we don't want to resolve the staticmethod. 

162 # Not using `cls.__dict__` because we want to check the entire MRO. 

163 cls 

164 and not isinstance( 

165 inspect.getattr_static(cls, name, default=None), staticmethod 

166 ) 

167 ): 

168 arg_names = arg_names[1:] 

169 # Remove any names that will be replaced with mocks. 

170 if hasattr(function, "__wrapped__"): 

171 arg_names = arg_names[num_mock_patch_args(function) :] 

172 return arg_names

195def ascii_escaped(val: bytes | str) -> str: 

196 r"""If val is pure ASCII, return it as an str, otherwise, escape 

197 bytes objects into a sequence of escaped bytes: 

198 

199 b'\xc3\xb4\xc5\xd6' -> r'\xc3\xb4\xc5\xd6' 

200 

201 and escapes strings into a sequence of escaped unicode ids, e.g.: 

202 

203 r'4\nV\U00043efa\x0eMXWB\x1e\u3028\u15fd\xcd\U0007d944' 

204 

205 Note: 

206 The obvious "v.decode('unicode-escape')" will return 

207 valid UTF-8 unicode if it finds them in bytes, but we 

208 want to return escaped bytes for any byte, even if they match 

209 a UTF-8 string. 

210 """ 

211 if isinstance(val, bytes): 

212 ret = val.decode("ascii", "backslashreplace") 

213 else: 

214 ret = val.encode("unicode_escape").decode("ascii") 

215 return ret.translate(_non_printable_ascii_translate_table)

260def get_user_id() -> int | None: 

261 """Return the current process's real user id or None if it could not be 

262 determined. 

263 

264 :return: The user id or None if it could not be determined. 

265 """ 

266 # mypy follows the version and platform checking expectation of PEP 484: 

267 # https://mypy.readthedocs.io/en/stable/common_issues.html?highlight=platform#python-version-and-system-platform-checks 

268 # Containment checks are too complex for mypy v1.5.0 and cause failure. 

269 if sys.platform == "win32" or sys.platform == "emscripten": 

270 # win32 does not have a getuid() function. 

271 # Emscripten has a return 0 stub. 

272 return None 

273 else: 

274 # On other platforms, a return value of -1 is assumed to indicate that 

275 # the current process's real user id could not be determined. 

276 ERROR = -1 

277 uid = os.getuid() 

278 return uid if uid != ERROR else None
