726 def _cmp_method(self, other, op): 

727 pc_func = ARROW_CMP_FUNCS[op.__name__] 

728 if isinstance(other, (ExtensionArray, np.ndarray, list)): 

729 try: 

730 boxed = self._box_pa(other) 

731 except pa.lib.ArrowInvalid: 

732 # e.g. GH#60228 [1, "b"] we have to operate pointwise 

733 res_values = [op(x, y) for x, y in zip(self, other)] 

734 result = pa.array(res_values, type=pa.bool_(), from_pandas=True) 

735 else: 

736 try: 

737 result = pc_func(self._pa_array, boxed) 

738 except pa.ArrowNotImplementedError: 

739 result = ops.invalid_comparison(self, other, op) 

740 result = pa.array(result, type=pa.bool_()) 

741 

742 elif is_scalar(other): 

743 try: 

744 result = pc_func(self._pa_array, self._box_pa(other)) 

745 except (pa.lib.ArrowNotImplementedError, pa.lib.ArrowInvalid): 

746 mask = isna(self) | isna(other) 

747 valid = ~mask 

748 result = np.zeros(len(self), dtype="bool") 

749 np_array = np.array(self) 

750 try: 

751 result[valid] = op(np_array[valid], other) 

752 except TypeError: 

753 result = ops.invalid_comparison(self, other, op) 

754 result = pa.array(result, type=pa.bool_()) 

755 result = pc.if_else(valid, result, None) 

756 else: 

757 raise NotImplementedError( 

758 f"{op.__name__} not implemented for {type(other)}" 

759 ) 

760 return ArrowExtensionArray(result)

772 def _evaluate_op_method(self, other, op, arrow_funcs) -> Self: 

773 pa_type = self._pa_array.type 

774 other_original = other 

775 other = self._box_pa(other) 

776 

777 if ( 

778 pa.types.is_string(pa_type) 

779 or pa.types.is_large_string(pa_type) 

780 or pa.types.is_binary(pa_type) 

781 ): 

782 if op in [operator.add, roperator.radd]: 

783 sep = pa.scalar("", type=pa_type) 

784 try: 

785 if op is operator.add: 

786 result = pc.binary_join_element_wise(self._pa_array, other, sep) 

787 elif op is roperator.radd: 

788 result = pc.binary_join_element_wise(other, self._pa_array, sep) 

789 except pa.ArrowNotImplementedError as err: 

790 raise TypeError( 

791 self._op_method_error_message(other_original, op) 

792 ) from err 

793 return type(self)(result) 

794 elif op in [operator.mul, roperator.rmul]: 

795 binary = self._pa_array 

796 integral = other 

797 if not pa.types.is_integer(integral.type): 

798 raise TypeError("Can only string multiply by an integer.") 

799 pa_integral = pc.if_else(pc.less(integral, 0), 0, integral) 

800 result = pc.binary_repeat(binary, pa_integral) 

801 return type(self)(result) 

802 elif ( 

803 pa.types.is_string(other.type) 

804 or pa.types.is_binary(other.type) 

805 or pa.types.is_large_string(other.type) 

806 ) and op in [operator.mul, roperator.rmul]: 

807 binary = other 

808 integral = self._pa_array 

809 if not pa.types.is_integer(integral.type): 

810 raise TypeError("Can only string multiply by an integer.") 

811 pa_integral = pc.if_else(pc.less(integral, 0), 0, integral) 

812 result = pc.binary_repeat(binary, pa_integral) 

813 return type(self)(result) 

814 if ( 

815 isinstance(other, pa.Scalar) 

816 and pc.is_null(other).as_py() 

817 and op.__name__ in ARROW_LOGICAL_FUNCS 

818 ): 

819 # pyarrow kleene ops require null to be typed 

820 other = other.cast(pa_type) 

821 

822 pc_func = arrow_funcs[op.__name__] 

823 if pc_func is NotImplemented: 

824 if pa.types.is_string(pa_type) or pa.types.is_large_string(pa_type): 

825 raise TypeError(self._op_method_error_message(other_original, op)) 

826 raise NotImplementedError(f"{op.__name__} not implemented.") 

827 

828 try: 

829 result = pc_func(self._pa_array, other) 

830 except pa.ArrowNotImplementedError as err: 

831 raise TypeError(self._op_method_error_message(other_original, op)) from err 

832 return type(self)(result)
