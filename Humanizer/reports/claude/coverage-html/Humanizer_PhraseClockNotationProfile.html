 533    /// <summary>
 534    /// Applies the Eifeler rule to a number-word value when enabled.
 535    /// The rule trims trailing 'n' from the last word of the resolved number value
 536    /// when the first word immediately after it starts with a non-blocking character.
 537    /// Non-number placeholder values are passed so the lookahead can resolve them inline.
 538    /// </summary>
 539    ReadOnlySpan<char> ApplyEifelerToValue(
 540        string value, string template, int afterIndex,
 541        string article, string nextArticle, string minuteSuffix, string dayPeriod)
 542    {
 543        if (!profile.ApplyEifelerRule || value.Length == 0)
 544        {
 545            return value;
 546        }
 547
 548        var nextWord = ExtractNextWordResolvingPlaceholders(template, afterIndex, article, nextArticle, minuteSuffix, da
 549        if (nextWord.Length == 0)
 550        {
 551            return value;
 552        }
 553
 554        // Apply the Eifeler rule to the last word of the resolved number value.
 555        var lastSpace = value.LastIndexOf(' ');
 556        if (lastSpace >= 0)
 557        {
 558            var lastWord = value[(lastSpace + 1)..];
 559            var eifeled = EifelerRule.ApplyIfNeeded(lastWord, nextWord);
 560            if (ReferenceEquals(eifeled, lastWord))
 561            {
 562                return value;
 563            }
 564
 565            return string.Concat(value.AsSpan(0, lastSpace + 1), eifeled);
 566        }
 567
 568        return EifelerRule.ApplyIfNeeded(value, nextWord);
 569    }

 571    /// <summary>
 572    /// Extracts the first word after the given position in the template.
 573    /// When a non-number placeholder is encountered, resolves it to its value and
 574    /// returns its first word, so the Eifeler rule sees the actual following word.
 575    /// </summary>
 576    static string ExtractNextWordResolvingPlaceholders(
 577        string template, int startIndex,
 578        string article, string nextArticle, string minuteSuffix, string dayPeriod)
 579    {
 580        var i = startIndex;
 581        while (i < template.Length)
 582        {
 583            // Skip spaces.
 584            while (i < template.Length && template[i] == ' ')
 585            {
 586                i++;
 587            }
 588
 589            if (i >= template.Length)
 590            {
 591                return "";
 592            }
 593
 594            if (template[i] == '{')
 595            {
 596                var closeBrace = template.IndexOf('}', i);
 597                if (closeBrace < 0)
 598                {
 599                    return "";
 600                }
 601
 602                var name = template.AsSpan(i + 1, closeBrace - i - 1);
 603
 604                // Resolve non-number placeholders to get their first word.
 605                var resolved = ResolveNonNumberPlaceholder(name, article, nextArticle, minuteSuffix, dayPeriod);
 606                if (resolved.Length > 0)
 607                {
 608                    // Return the first word of the resolved value.
 609                    var spaceIdx = resolved.IndexOf(' ');
 610                    return spaceIdx >= 0 ? resolved[..spaceIdx] : resolved;
 611                }
 612
 613                // Skip number placeholders and continue looking.
 614                i = closeBrace + 1;
 615                continue;
 616            }
 617
 618            // Extract the literal word until space or placeholder.
 619            var wordStart = i;
 620            while (i < template.Length && template[i] != ' ' && template[i] != '{')
 621            {
 622                i++;
 623            }
 624
 625            return template[wordStart..i];
 626        }
 627
 628        return "";
 629    }

 674    string ApplyDayPeriod(string basePhrase, int hour, bool usesNextHour)
 675    {
 676        if (profile.EarlyMorning.Length == 0 && profile.Morning.Length == 0 &&
 677            profile.Afternoon.Length == 0 && profile.Night.Length == 0)
 678        {
 679            return basePhrase;
 680        }
 681
 682        // For relative-hour style locales (ca, es), templates that reference {nextHour}
 683        // mean the phrasing is relative to the next hour (e.g., "twenty to five"),
 684        // so the day period should be based on the next hour's time slot.
 685        // For locales that say "current hour + minutes" (ar, fa, ku), the period
 686        // must always reflect the current hour.
 687        var periodHour = usesNextHour ? hour + 1 : hour;
 688        var period = GetDayPeriod(periodHour);
 689        if (period.Length == 0)
 690        {
 691            return basePhrase;
 692        }
 693
 694        return profile.DayPeriodPosition == PhraseClockDayPeriodPosition.Prefix
 695            ? period + " " + basePhrase
 696            : basePhrase + " " + period;
 697    }

 699    string GetDayPeriod(int hour)
 700    {
 701        var normalized = hour % 24;
 702        if (normalized is >= 1 and < 6)
 703        {
 704            return profile.EarlyMorning;
 705        }
 706
 707        if (normalized is >= 6 and < 12)
 708        {
 709            return profile.Morning;
 710        }
 711
 712        if (normalized is >= 12 and < 21)
 713        {
 714            return profile.Afternoon;
 715        }
 716
 717        return profile.Night;
 718    }
